# One executable per test file, each registered with CTest.
function(resindeck_test name)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE resindeck_core resindeck_warnings Qt6::Test)
  add_test(NAME ${name} COMMAND ${name})
endfunction()

resindeck_test(test_des)
resindeck_test(test_status)
resindeck_test(test_halot)
resindeck_test(test_halot_driver)
resindeck_test(test_config)
resindeck_test(test_cxdlp)
resindeck_test(test_lan)
resindeck_test(test_mesh)
resindeck_test(test_printer)
resindeck_test(test_server)
target_compile_definitions(test_server PRIVATE RESINDECK_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
resindeck_test(test_cli)
target_compile_definitions(test_cli PRIVATE RESINDECK_CLI="$<TARGET_FILE:resindeck-cli>")
add_dependencies(test_cli resindeck-cli)
